git add .

# 3. 정리된 상태를 깃허브에 최종 반영합니다.
git commit -m "Fix: Clean up venv and pycache from tracking"
git push


# 1. 깃허브 주소를 원격 저장소(origin)로 등록합니다. (질문하신 내용 실행)
